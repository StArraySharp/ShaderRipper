// One-off diagnostic: print every pass's stencil slots for one shader, so the
// Front/Back merge decision can be made against real data instead of guesses.
using System;
using System.IO;
using System.Linq;

namespace ShaderRip
{
    internal static class StencilProbe
    {
        private static int Main(string[] args)
        {
            string dataDir = args.Length > 0 ? args[0]
                : @"E:\SteamLibrary\steamapps\common\A Dance of Fire and Ice\A Dance of Fire and Ice_Data";
            string filter = args.Length > 1 ? args[1] : "zWriteUI";

            string schema = Path.Combine(AppContext.BaseDirectory, "schema", "shader_typetypes_6000.json");
            if (!File.Exists(schema))
                schema = Path.GetFullPath(Path.Combine(AppContext.BaseDirectory, "..", "..", "..", "schema", "shader_typetypes_6000.json"));
            var root = TypeNode.LoadSchema(schema);
            var sf = SerializedFile.Load(Path.Combine(dataDir, "resources.assets"), root);

            foreach (var sh in sf.Shaders)
            {
                string name = sh.ParsedForm?.Name ?? sh.Name ?? "";
                if (!name.Contains(filter)) continue;
                Console.WriteLine($"=== {name} pathID={sh.PathId}");
                foreach (var ss in sh.ParsedForm.SubShaders)
                    foreach (var ps in ss.Passes)
                    {
                        Console.WriteLine($"  pass '{ps.StateName}'  Ref={ps.StencilRef} Read={ps.StencilReadMask} Write={ps.StencilWriteMask}  ops={ps.StencilOps.Count}");
                        for (int i = 0; i < ps.StencilOps.Count; i++)
                        {
                            var o = ps.StencilOps[i];
                            Console.WriteLine($"    [{i}] Comp={o.Comp} Pass={o.PassOp} Fail={o.Fail} ZFail={o.ZFail}");
                        }
                        foreach (var b in ps.Blends)
                            Console.WriteLine($"    blend Src={b.SrcBlend} Dst={b.DestBlend} ColMask={b.ColMask}");
                    }
            }
            return 0;
        }
    }
}
